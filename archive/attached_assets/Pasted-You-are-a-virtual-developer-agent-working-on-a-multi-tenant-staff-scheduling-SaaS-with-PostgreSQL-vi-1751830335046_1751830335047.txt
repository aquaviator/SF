2. **Admin API Endpoints**

     ```ts
     // GET /api/admin/landing-page
     app.get('/api/admin/landing-page', adminAuth, async (req, res) => {
       const page = await db.select().from(landingPages).where(eq(landingPages.tenantId, req.user.tenantId)).get();
       res.json(page || { layout: [] });
     });
     // PATCH /api/admin/landing-page
     app.patch('/api/admin/landing-page', adminAuth, async (req, res) => {
       const { layout } = req.body;
       const upsert = await db
         .insert(landingPages)
         .values({ tenantId: req.user.tenantId, layout })
         .onConflictDoUpdate({ target: landingPages.tenantId, set: { layout, updatedAt: sql`now()` } });
       res.json(upsert);
     });
     ```

import speakeasy from 'speakeasy';
import qrcode from 'qrcode';
import { nanoid } from 'nanoid';
// POST /api/admin/2fa/setup
app.post('/api/admin/2fa/setup', adminAuth, async (req, res) => {
  const secret = speakeasy.generateSecret({ name: `ShiftFlo (${req.user.username})` });
  await db.update(siteAdmins).set({ totp_secret: secret.base32 }).where(eq(siteAdmins.id, req.user.id));
  const qr = await qrcode.toDataURL(secret.otpauth_url);
  res.json({ otpauth_url: secret.otpauth_url, qr });
});
// POST /api/admin/2fa/verify
app.post('/api/admin/2fa/verify', adminAuth, async (req, res) => {
  const admin = await db.select().from(siteAdmins).where(eq(siteAdmins.id, req.user.id)).get();
  const valid = speakeasy.totp.verify({ secret: admin.totp_secret, encoding: 'base32', token: req.body.token, window: 1 });
  if (!valid) return res.status(400).json({ message: 'Invalid code' });
  const codes = Array.from({ length: 10 }, () => nanoid(8));
  await db.update(siteAdmins).set({ is_2fa_enabled: true, recovery_codes: codes }).where(eq(siteAdmins.id, req.user.id));
  res.json({ recoveryCodes: codes });
});
// POST /api/admin/2fa/challenge
app.post('/api/admin/2fa/challenge', async (req, res) => {
  const session = req.session; // assume pending_2fa session
  const admin = await db.select().from(siteAdmins).where(eq(siteAdmins.id, session.userId)).get();
  const { token } = req.body;
  const validTotp = speakeasy.totp.verify({ secret: admin.totp_secret, encoding: 'base32', token, window: 1 });
  let valid = validTotp;
  if (!valid && admin.recovery_codes.includes(token)) {
    valid = true;
    await db.update(siteAdmins)
      .set({ recovery_codes: sql`${siteAdmins.recovery_codes} - ${token}` })
      .where(eq(siteAdmins.id, admin.id));
  }
  if (!valid) return res.status(400).json({ message: 'Invalid code' });
  session.is2faVerified = true;
  res.json({ success: true });
});

import { useState, useEffect } from 'react';
export default function TwoFASetup() {
  const [qr, setQr] = useState('');
  const [token, setToken] = useState('');
  const [codes, setCodes] = useState([]);
  useEffect(() => {
    fetch('/api/admin/2fa/setup', { method: 'POST', credentials: 'include' })
      .then(r => r.json()).then(data => setQr(data.qr));
  }, []);
  const verify = () => fetch('/api/admin/2fa/verify', {
    method: 'POST', credentials: 'include',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify({ token })
  }).then(r => r.json()).then(d => setCodes(d.recoveryCodes));
  return (
    <div>
      <h2>Set Up 2FA</h2>
      {qr && <img src={qr} alt="Scan this QR with your Authenticator app" />}
      <input value={token} onChange={e => setToken(e.target.value)} placeholder="Enter code" />
      <button onClick={verify}>Enable 2FA</button>
      {codes.length > 0 && (
        <div>
          <h3>Recovery Codes</h3>
          <ul>{codes.map(c => <li key={c}>{c}</li>)}</ul>
        </div>
      )}
    </div>
  );
}

import { useState } from 'react';

export default function TwoFAChallenge() {
  const [token, setToken] = useState('');
  const nav = useNavigate();
  const submit = () => fetch('/api/admin/2fa/challenge', {
    method: 'POST', credentials: 'include',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify({ token })
  }).then(r => r.ok && nav('/admin'));
  return (
    <div>
      <h2>Enter 2FA Code or Recovery Code</h2>
      <input value={token} onChange={e => setToken(e.target.value)} placeholder="6-digit code or recovery code" />
      <button onClick={submit}>Verify</button>
    </div>
  );
}
