ts

const handleSwap = async () => {
  const allowed = checkSwapPolicyStatus(shift);
  console.log('REQUEST_SWAP clicked', { shift, allowed });
  if (!allowed) return escalateToManager('swap', shift);
  try {
    await requestSwap(shift.id);
    console.log('REQUEST_SWAP success', { shift });
    refreshShifts();
  } catch (err) {
    console.error('REQUEST_SWAP error', err);
    showError('Could not request swap');
  }
};

const handleCancel = async () => {
  const allowed = checkCancelPolicyStatus(shift);
  console.log('REQUEST_CANCEL clicked', { shift, allowed });
  if (!allowed) return escalateToManager('cancel', shift);
  try {
    await requestCancel(shift.id);
    console.log('REQUEST_CANCEL success', { shift });
    refreshShifts();
  } catch (err) {
    console.error('REQUEST_CANCEL error', err);
    showError('Could not request cancellation');
  }
};
